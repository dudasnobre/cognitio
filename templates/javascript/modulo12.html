12. Projeto Final: Aplicação Interativa

Hora de juntar tudo: DOM, eventos, arrays, localStorage e JavaScript moderno em um projeto clássico e extremamente comum no mercado — uma lista de tarefas (To-Do List) funcional e persistente.

- O Desafio

  • ✅ Um formulário para adicionar novas tarefas
  • ✅ Marcar uma tarefa como concluída (clique para alternar)
  • ✅ Remover uma tarefa da lista
  • ✅ Salvar tudo no localStorage, para não perder ao recarregar a página
  • ✅ Mostrar quantas tarefas restam

- Estrutura de Dados

let tarefas = JSON.parse(localStorage.getItem('tarefas') || '[]');
// Cada tarefa: { id: 1, texto: 'Estudar JS', concluida: false }

- Funções Principais

function adicionarTarefa(texto) {
    tarefas.push({ id: Date.now(), texto, concluida: false });
    salvar();
    renderizar();
}

function alternarTarefa(id) {
    tarefas = tarefas.map(t =>
        t.id === id ? { ...t, concluida: !t.concluida } : t
    );
    salvar();
    renderizar();
}

function removerTarefa(id) {
    tarefas = tarefas.filter(t => t.id !== id);
    salvar();
    renderizar();
}

function salvar() {
    localStorage.setItem('tarefas', JSON.stringify(tarefas));
}

- Renderizando a Lista

function renderizar() {
    const lista = document.querySelector('#lista-tarefas');
    lista.innerHTML = '';
    tarefas.forEach(t => {
        const li = document.createElement('li');
        li.textContent = t.texto;
        li.style.textDecoration = t.concluida ? 'line-through' : 'none';
        li.addEventListener('click', () => alternarTarefa(t.id));
        lista.appendChild(li);
    });
    document.querySelector('#contador').textContent =
        `${tarefas.filter(t => !t.concluida).length} tarefa(s) pendente(s)`;
}
renderizar(); // chamada inicial, ao carregar a página
Este é o padrão de toda aplicação com estado
Estado (o array tarefas) → funções que alteram o estado → uma função renderizar() que redesenha a tela a partir do estado atual. Esse ciclo é a base conceitual de frameworks como React e Vue — você já entende a ideia central!
